function readAllRows_(sheet) {
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  var headers = data[0];
  var rows = [];
  for (var i = 1; i < data.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) obj[headers[j]] = data[i][j];
    rows.push(obj);
  }
  return rows;
}

function jsonOut_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  return jsonOut_({ status: "ok", serverTime: new Date().toISOString(), rows: readAllRows_(sheet) });
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var body = JSON.parse(e.postData.contents);
    var action = body.action || "upsert";

    if (action === "delete") {
      if (sheet.getLastRow() > 1) {
        var allData = sheet.getDataRange().getValues();
        var headers = allData[0];
        var uidCol = headers.indexOf("uid");
        var hsbaCol = headers.indexOf("ma_hsba");
        var targetHsba = String(body.ma_hsba || "").trim().toLowerCase();
        for (var r = allData.length - 1; r >= 1; r--) {
          if ((uidCol !== -1 && body.uid && allData[r][uidCol] === body.uid) ||
              (hsbaCol !== -1 && targetHsba && String(allData[r][hsbaCol]).trim().toLowerCase() === targetHsba)) {
            sheet.deleteRow(r + 1);
            break;
          }
        }
      }
      return jsonOut_({ status: "ok", serverTime: new Date().toISOString(), rows: readAllRows_(sheet) });
    }

    if (action === "claim" || action === "release") {
      var headers = sheet.getLastRow() === 0 ? [] : sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
      ["uid", "ma_hsba", "editing_by", "editing_at"].forEach(function(k) {
        if (headers.indexOf(k) === -1) headers.push(k);
      });
      if (sheet.getLastRow() === 0) sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
      else if (headers.length > sheet.getLastColumn()) sheet.getRange(1, 1, 1, headers.length).setValues([headers]);

      var allData = sheet.getDataRange().getValues();
      var uidCol = headers.indexOf("uid");
      var hsbaCol = headers.indexOf("ma_hsba");
      var edByCol = headers.indexOf("editing_by");
      var edAtCol = headers.indexOf("editing_at");
      var targetHsba = String(body.ma_hsba || "").trim().toLowerCase();
      for (var r = 1; r < allData.length; r++) {
        if ((uidCol !== -1 && allData[r][uidCol] === body.uid) ||
            (hsbaCol !== -1 && targetHsba && String(allData[r][hsbaCol]).trim().toLowerCase() === targetHsba)) {
          sheet.getRange(r + 1, edByCol + 1).setValue(action === "claim" ? (body.user || "") : "");
          sheet.getRange(r + 1, edAtCol + 1).setValue(action === "claim" ? new Date().toISOString() : "");
          break;
        }
      }
      return jsonOut_({ status: "ok", serverTime: new Date().toISOString(), rows: readAllRows_(sheet) });
    }

    var incomingRows = action === "bulk_upsert" ? (body.rows || []) : [body.row];
    if (body.imgBase64 && incomingRows[0]) {
      var folder = DriveApp.getFoldersByName("CTG_ACOG2025_Images").hasNext()
        ? DriveApp.getFoldersByName("CTG_ACOG2025_Images").next()
        : DriveApp.createFolder("CTG_ACOG2025_Images");
      var fileName = (incomingRows[0].ma_hsba || incomingRows[0].ma_nc || "CTG") + ".jpg";
      var blob = Utilities.newBlob(Utilities.base64Decode(body.imgBase64.split(",")[1]), "image/jpeg", fileName);
      var file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      incomingRows[0].link_anh_drive = file.getUrl();
    }

    var headers = sheet.getLastRow() === 0 ? [] : sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    var headerChanged = false;
    incomingRows.forEach(function(rowObj) {
      Object.keys(rowObj).forEach(function(k) {
        if (headers.indexOf(k) === -1) { headers.push(k); headerChanged = true; }
      });
    });
    if (sheet.getLastRow() === 0 || headerChanged) {
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    }

    var allData = sheet.getDataRange().getValues();
    var uidCol = headers.indexOf("uid");
    var hsbaCol = headers.indexOf("ma_hsba");
    var forceFields = ["editing_by","editing_at","ctg_verified","qa_10pct","final_locked","screen_status","screen_reason","screen_note","doc_ctg_group","has_risk","risk_detail","found_via_paper","trang_thai_nhap","cap_nhat_luc","last_edited_by"];

    incomingRows.forEach(function(rowObj) {
      var targetRow = -1;
      var normHsba = String(rowObj.ma_hsba || "").trim().toLowerCase();
      for (var r = 1; r < allData.length; r++) {
        if ((uidCol !== -1 && rowObj.uid && allData[r][uidCol] === rowObj.uid) ||
            (hsbaCol !== -1 && normHsba && String(allData[r][hsbaCol]).trim().toLowerCase() === normHsba)) {
          targetRow = r + 1; break;
        }
      }
      if (targetRow > 0) {
        var existingRow = allData[targetRow - 1];
        var mergedValues = headers.map(function(h, idx) {
          var incomingVal = rowObj[h];
          var oldVal = existingRow[idx];
          if (forceFields.indexOf(h) !== -1) {
            return incomingVal !== undefined ? incomingVal : (oldVal !== undefined ? oldVal : "");
          }
          if (incomingVal !== undefined && incomingVal !== "" && incomingVal !== null) return incomingVal;
          return oldVal !== undefined ? oldVal : "";
        });
        sheet.getRange(targetRow, 1, 1, mergedValues.length).setValues([mergedValues]);
        allData[targetRow - 1] = mergedValues;
      } else {
        var newValues = headers.map(function(h) { return rowObj[h] !== undefined ? rowObj[h] : ""; });
        sheet.appendRow(newValues);
        allData.push(newValues);
      }
    });

    return jsonOut_({ status: "ok", serverTime: new Date().toISOString(), rows: readAllRows_(sheet) });
  } finally {
    lock.releaseLock();
  }
}

📖 Hướng dẫn Nhập liệu — TRANG ĐOÀN

Bước 1: Nhập tên của bạn để tránh làm trùng ca

Bấm nút "👤 Tên của tôi" ở góc trên cùng bên phải → gõ tên của bạn (VD: Mai, Lan, Hùng...) → bấm "💾 Lưu tên của tôi". Khi bạn mở một bệnh án để nhập, màn hình của các bạn khác sẽ tự động hiện nhãn 🔒 Tên bạn đang nhập để mọi người tránh chọn trùng ca đó.

Bước 2: Chọn ca đang rảnh để nhập liệu

  • Ở ô lọc trạng thái, chọn "🆓 Ca đang rảnh (Chưa ai mở & Chưa xong)" hoặc tìm các dòng có nhãn 🆕 Mới duyệt — Chờ nhập.
  • Không mở các dòng đang có nhãn 🔒 [Tên bạn khác] đang nhập hoặc 🔐 Đã chốt (chế độ chỉ đọc).

Bước 3: Nhập 7 phần A – G & Chụp ảnh CTG

  • Các ô có nền vàng nhạt là biến số bắt buộc. Tuổi mẹ (A1), BMI (A7), Nhóm tuổi thai (D3), Thời gian 30p (F11) sẽ tự động tính toán.
  • Nếu chụp/up ảnh CTG: Sang tab B. CTG → bấm "📷 Chụp / Chọn ảnh CTG" (đoạn ghi ≥ 30 phút rõ nét) → hệ thống điền gợi ý thông số ban đầu → kiểm tra lại, sửa chỗ chưa đúng rồi bấm "💾 Lưu & Đồng bộ nhóm".

Bước 4: Khi không có mạng (Offline)

Bạn cứ nhập và bấm "💾 Lưu & Đồng bộ nhóm" bình thường. App tự lưu an toàn trên điện thoại của bạn và sẽ tự động đồng bộ lên kho chung của nhóm ngay khi có Wi-Fi / 4G trở lại.

Bước 5: Cài app ra màn hình chính điện thoại (Làm 1 lần)

  • Trên iPhone / iPad (Safari): Mở link bằng Safari → bấm nút Chia sẻ (ô vuông có mũi tên lên ở dưới cùng) → chọn "Thêm vào MH chính" (Add to Home Screen) → bấm Thêm.
  • Trên Android (Chrome): Mở link bằng Chrome → bấm nút 3 chấm (⋮) góc trên phải → chọn "Thêm vào màn hình chính" / "Cài đặt ứng dụng".
